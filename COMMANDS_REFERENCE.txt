# Video Watermarker - Command Reference

## Main Watermarking Commands

# Rough output directory
python video_watermarker.py --input-dir {INPUT_DIR} --output-dir {OUTPUT_DIR} --logo-path "{LOGO_PATH}" --position "top-right" --scale 0.15 --transparency 150

# Video output directory
python video_watermarker.py --input-dir {INPUT_DIR} --output-dir {OUTPUT_DIR} --logo-path "{LOGO_PATH}" --position "top-right" --scale 0.15 --transparency 150

# Temp video directory
python video_watermarker.py --input-dir {INPUT_DIR} --output-dir {OUTPUT_DIR} --logo-path "{LOGO_PATH}" --position "top-right" --scale 0.15 --transparency 150

# Alternative logo path
python video_watermarker.py --input-dir {INPUT_DIR} --output-dir {OUTPUT_DIR} --logo-path "{LOGO_PATH}" --position "top-right" --scale 0.15 --transparency 150

## Parameters
# --position: "top-right", "top-left", "bottom-right", "bottom-left", "center"
# --scale: 0.1 to 1.0 (size of watermark relative to video)
# --transparency: 0 (fully transparent) to 255 (fully opaque)
